=== ШАГ 1: Практические примеры: создание узлов (CREATE) ===

neo4j> MATCH (n) DETACH DELETE n;

neo4j> CREATE (a:Person {name: "Алиса", age: 30}) RETURN a;
a
(:Person {name: "Алиса", age: 30})

neo4j> CREATE (b:Person {name: "Боб", age: 25}), (c:Person {name: "Кэрол", age: 28}) RETURN b, c;
b, c
(:Person {name: "Боб", age: 25}), (:Person {name: "Кэрол", age: 28})

neo4j> MATCH (p:Person) RETURN p.name, p.age;
p.name, p.age
"Алиса", 30
"Боб", 25
"Кэрол", 28
