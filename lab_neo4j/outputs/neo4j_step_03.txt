=== ШАГ 3: Практические примеры: изменение и удаление узлов (SET, REMOVE, DELETE) ===

neo4j> MATCH (p:Person {name: "Боб"}) SET p.city = "Москва" RETURN p;
p
(:Person {name: "Боб", city: "Москва", age: 25})

neo4j> MATCH (p:Person {name: "Кэрол"}) SET p.age = p.age + 1 RETURN p.name, p.age;
p.name, p.age
"Кэрол", 29

neo4j> MATCH (p:Person {name: "Боб"}) REMOVE p.city RETURN p;
p
(:Person {name: "Боб", age: 25})

neo4j> MATCH (p:Person {name: "Кэрол"}) DELETE p;

neo4j> MATCH (p:Person) RETURN count(p) AS осталось_персон;
осталось_персон
2
