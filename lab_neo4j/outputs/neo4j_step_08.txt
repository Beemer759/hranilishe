neo4j> MATCH (p:Person) RETURN count(p) AS персонажей;
персонажей
12

neo4j> MATCH ()-[r]->() RETURN count(r) AS связей;
связей
37

"Ирина Волкова", 6
"Егор Морозов", 5
"Марк Волков", 5
"Кирилл Попов", 5
"Глеб Орлов", 5
"Жанна Лебедева", 4
"Вера Крамер", 4

neo4j> MATCH (a:Person {name:"Борис Крамер"})-[:FRIEND]-(f:Person) RETURN f.name AS друг;
друг
"Кирилл Попов"
"Анна Соколова"

neo4j> MATCH (a:Person)-[:ENEMY]->(b:Person) RETURN a.name AS кто, b.name AS против_кого;
кто, против_кого
"Егор Морозов", "Борис Крамер"
"Егор Морозов", "Анна Соколова"
"Жанна Лебедева", "Анна Соколова"

neo4j> MATCH (p:Person)-[:MEMBER_OF]->(g:Group) RETURN g.name AS группа, collect(p.name) AS участники;
группа, участники
"Семья Крамер", ["Ирина Волкова", "Вера Крамер", "Борис Крамер"]
"Клуб «Полночь»", ["Глеб Орлов", "Марк Волков", "Захар Титов"]
"Следственная группа", ["Анна Соколова", "Кирилл Попов", "Борис Крамер"]

neo4j> MATCH (p:Person)-[:LOVES]->(q:Person) RETURN p.name AS влюблён, q.name AS в_кого;
влюблён, в_кого
"Марк Волков", "Дина Орлова"
"Захар Титов", "Ирина Волкова"
"Кирилл Попов", "Дина Орлова"

neo4j> MATCH path = shortestPath((a:Person {name:"Анна Соколова"})-[*]-(b:Person {name:"Дина Орлова"})) RETURN [n IN nodes(path) | n.name] AS путь, length(path) AS длина;
путь, длина
["Анна Соколова", "Марк Волков", "Дина Орлова"], 2

neo4j> MATCH (p:Person) WHERE p.age >= 40 RETURN p.name AS имя, p.role AS роль, p.age AS возраст ORDER BY p.age DESC;
