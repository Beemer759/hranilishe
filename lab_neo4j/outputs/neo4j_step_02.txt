=== ШАГ 2: Практические примеры: чтение данных (MATCH, WHERE, агрегаты) ===

neo4j> MATCH (p:Person) WHERE p.age > 26 RETURN p.name AS имя, p.age AS возраст ORDER BY p.age DESC;
имя, возраст
"Алиса", 30
"Кэрол", 28

neo4j> MATCH (p:Person {name: "Алиса"}) RETURN p;
p
(:Person {name: "Алиса", age: 30})

neo4j> MATCH (p:Person) RETURN count(p) AS всего_персон;
всего_персон
3

neo4j> MATCH (p:Person) RETURN avg(p.age) AS средний_возраст, max(p.age) AS максимум;
средний_возраст, максимум
27.666666666666668, 30
