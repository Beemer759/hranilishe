=== ШАГ 7: Изменение документов: $set, $inc, $push и $unset ===

sandbox> db.humans.updateOne({name: "John"}, {$set: {city: "Moscow"}})
{
  acknowledged: true,
  insertedId: null,
  matchedCount: 1,
  modifiedCount: 1,
  upsertedCount: 0
}

sandbox> db.humans.updateMany({}, {$inc: {age: 1}})
{
  acknowledged: true,
  insertedId: null,
  matchedCount: 5,
  modifiedCount: 5,
  upsertedCount: 0
}

sandbox> db.humans.find({}, {name: 1, age: 1, city: 1, _id: 0})
[
  { name: 'John', age: 45, city: 'Moscow' },
  { name: 'Test', age: 36 },
  { name: 'Ann', age: 26 },
  { name: 'Mary', age: 23 },
  { name: 'Peter', age: 40 }
]

sandbox> db.humans.updateOne({name: "Mary"}, {$unset: {age: ""}})
{
  acknowledged: true,
  insertedId: null,
  matchedCount: 1,
  modifiedCount: 1,
  upsertedCount: 0
}

sportsmen> db.sportsmen.updateOne({name: "Dominik Hašek"}, {$push: {clubs: {clubName: "HC Torino", firstYear: 2000, lastYear: 2001}}})
{
  acknowledged: true,
  insertedId: null,
  matchedCount: 1,
  modifiedCount: 1,
  upsertedCount: 0
}

sportsmen> db.sportsmen.find({name: "Dominik Hašek"}, {name: 1, "clubs.clubName": 1, _id: 0})
